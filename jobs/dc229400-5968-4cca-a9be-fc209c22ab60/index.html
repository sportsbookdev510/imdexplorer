/// Anyone can block the owner's permanent renouncement of a farm's emissions-manager role by
/// putting 1 wei into the reward reserve between the owner's `returnUncommittedRewards` and
/// `renounceEmissionsManager` transactions (front-run), via `fundRewards` or a plain transfer
/// followed by the permissionless `syncRewardReserve`.
contract RenounceGriefTest is Test {
    address internal constant OWNER = address(0xDEA11b3A70b77134424d436f6eb455f1Abbd4449);
    address internal constant TREASURY = address(0x7EA5);
    address internal constant MALLORY = address(0xBAD);

    LaunchToken internal locker;
    LockerFarm internal farm;

    function setUp() public {
        vm.warp(1_700_000_000);
        vm.prank(OWNER);
        locker = new LaunchToken();
        farm = new LockerFarm(address(locker), address(locker), TREASURY, OWNER);
        vm.startPrank(OWNER);
        locker.approve(address(farm), 1_000 ether);
        farm.fundRewards(1_000 ether);
        locker.transfer(MALLORY, 2);
        vm.stopPrank();
    }

    function test_strangerCannotBlockRenounceViaFundRewards() public {
        uint256 reserve = farm.rewardReserve();
        vm.prank(OWNER);
        farm.returnUncommittedRewards(reserve);
        assertEq(farm.rewardReserve(), 0);

        // Front-run of the renounce transaction.
        vm.startPrank(MALLORY);
        locker.approve(address(farm), 1);
        farm.fundRewards(1);
        vm.stopPrank();

        vm.prank(OWNER);
        farm.renounceEmissionsManager();
        assertEq(farm.emissionsManager(), address(0), "owner could not renounce");
    }

    function test_strangerCannotBlockRenounceViaDonationAndSync() public {
        uint256 reserve = farm.rewardReserve();
        vm.prank(OWNER);
        farm.returnUncommittedRewards(reserve);
        assertEq(farm.rewardReserve(), 0);

        vm.startPrank(MALLORY);
        locker.transfer(address(farm), 1);
        farm.syncRewardReserve();
        vm.stopPrank();

        vm.prank(OWNER);
        farm.renounceEmissionsManager();
        assertEq(farm.emissionsManager(), address(0), "owner could not renounce");
    }
}
